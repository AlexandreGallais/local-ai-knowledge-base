# Sample XML file: Numerical data (LINQ to XML)

  <TaxRate>7.25</TaxRate>
  <Data>
    <Category>A</Category>
    <Quantity>3</Quantity>

  <Data>

    <Price>89.99</Price>
  </Data>
  <Data>

  </Data>
  <Data>
    <Category>A</Category>

  </Data>
  <Data>
    <Category>B</Category>
    <Quantity>8</Quantity>
    <Price>6.99</Price>
  </Data>
</Root>
